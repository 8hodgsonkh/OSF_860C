heres a short list of the main tweaks ive done to make this lil beast into a power house. i always wondered. why doesnt a low gear actually feel torquey at all? after increasing amps and doing a lot of tweaks, i stil felt like i was missing that final "punch" turns out, duty cycle was the perfect solution.
suprisingly motor temp hasnt really increased whatsoever
faster time to a efficient rpm  = less time building heat ¯\_(ツ)_/¯
motor also handles increase amps like a champ. obviously dont ride in max gear, uphill, throttle only, because it will pull, and it will get hot then
these are fun performance tweaks but used sensibly shouldnt increase wear too much as theres some safe guards here. but it all depends how you treat your bike!
Duty cycle - instant torque and hard start with soft start ntil load detected to help stop that instant snap of your chain and casette - before this it was too ruthless and adding a soft start didnt take away any fun at all
times are tweakable if u want to change it a bit

 {
            /* ---------------- Soft-launch until load is detected ---------------- */
            /* C89-safe statics (persist across IRQ calls) */
            static uint8_t  soft_launch_active  = 0u;
            static uint16_t soft_launch_counter = 0u;

            /* Tunables */
            const uint16_t SOFT_LAUNCH_MAX_TICKS = 300u;     /* timeout failsafe */
            const uint8_t  SOFT_START_DUTY       = (uint8_t)(PWM_DUTY_CYCLE_STARTUP / 2u);  /* gentle spin */
            const uint8_t  SOFT_CEILING_MAX      = 80u;      /* max duty during preload (0..255) */
            const uint8_t  SOFT_CEILING_STEP_N   = 4u;       /* +1 to ceiling every N ticks */

            /* Convert ≈3 A to 10-bit ADC steps (current filter is in ADC steps) */
            uint16_t I_THRESH_ADC10_u16 =
            (uint16_t)(300u / (uint16_t)BATTERY_CURRENT_PER_10_BIT_ADC_STEP_X100);
            if (I_THRESH_ADC10_u16 == 0u)   I_THRESH_ADC10_u16 = 1u;
            if (I_THRESH_ADC10_u16 > 255u)  I_THRESH_ADC10_u16 = 255u;
            const uint16_t I_THRESH_ADC10 = I_THRESH_ADC10_u16;

            /* We clamp the *target* during soft-launch, not the actual duty. */
            uint8_t target_for_ramp = ui8_controller_duty_cycle_target;

            /* Enter soft-launch when starting from zero, target > 0, and no load yet */
            if (soft_launch_active == 0u) {
                if ((ui8_g_duty_cycle == 0u) &&
                    (ui8_controller_duty_cycle_target > 0u) &&
                    (ui16_adc_battery_current_filtered < I_THRESH_ADC10))
                {
                    soft_launch_active  = 1u;
                    soft_launch_counter = 0u;
                }
            } else {
                /* Raise a soft duty ceiling slowly while waiting for load to appear */
                uint8_t soft_ceiling = SOFT_START_DUTY;
                soft_ceiling = (uint8_t)(soft_ceiling + (uint8_t)(soft_launch_counter / SOFT_CEILING_STEP_N));
                if (soft_ceiling > SOFT_CEILING_MAX) soft_ceiling = SOFT_CEILING_MAX;

                if (target_for_ramp > soft_ceiling)
                    target_for_ramp = soft_ceiling;

                /* keep time and exit when load shows up or we time out */
                soft_launch_counter++;
                if ((ui16_adc_battery_current_filtered >= I_THRESH_ADC10) ||
                    (soft_launch_counter >= SOFT_LAUNCH_MAX_TICKS))
                {
                    soft_launch_active = 0u;
                }
            }

            /* ---------------- Snappy but safe duty ramp (uses target_for_ramp) ---------------- */
            const uint8_t FAST_GAP_UP    = 8u;   /* if target-actual >= 8, fast ramp */
            const uint8_t STEP_UP_FAST   = 3u;   /* +3 when far from target */
            const uint8_t STEP_UP_SLOW   = 1u;   /* +1 when close */
            const uint8_t STEP_DOWN_FAST = 2u;   /* -2 when far over / braking / over-phase */
            const uint8_t STEP_DOWN_SLOW = 1u;   /* -1 gentle */

            /* Direction flags (no stdbool) */
            uint8_t must_ramp_down = 0u;
            uint8_t can_ramp_up    = 0u;

            if ((target_for_ramp < ui8_g_duty_cycle) ||                                          /* target < actual */
                (ui16_controller_adc_battery_current_target < ui16_adc_battery_current_filtered) || /* requested Ibat < measured */
                (ui16_adc_motor_phase_current > ui16_adc_motor_phase_current_max) ||             /* phase over limit */
                /* (ui16_hall_counter_total < (HALL_COUNTER_FREQ / MOTOR_OVER_SPEED_ERPS)) || */ /* overspeed (disabled) */
                (ui16_adc_voltage < ui16_adc_voltage_cut_off) ||                                 /* undervoltage */
                (ui8_brake_state != 0))
            {
                must_ramp_down = 1u;
            }
            else if ((target_for_ramp > ui8_g_duty_cycle) &&
                (ui16_controller_adc_battery_current_target > ui16_adc_battery_current_filtered))
            {
                can_ramp_up = 1u;
            }

            if (must_ramp_down) {
                ui8_counter_duty_cycle_ramp_up = 0;

                if (++ui8_counter_duty_cycle_ramp_down > ui8_controller_duty_cycle_ramp_down_inverse_step) {
                    ui8_counter_duty_cycle_ramp_down = 0;

                    /* FW first, then duty—keeps original behavior */
                    if (ui8_fw_hall_counter_offset > 0) {
                        ui8_fw_hall_counter_offset--;
                    } else if (ui8_g_duty_cycle > 0) {
                        uint8_t step = STEP_DOWN_SLOW;
                        if ((ui8_g_duty_cycle > (uint8_t)(target_for_ramp + FAST_GAP_UP)) ||
                            (ui8_brake_state != 0) ||
                            (ui16_adc_motor_phase_current > ui16_adc_motor_phase_current_max))
                        {
                            step = STEP_DOWN_FAST;
                        }
                        ui8_g_duty_cycle = (ui8_g_duty_cycle > step) ? (uint8_t)(ui8_g_duty_cycle - step) : 0u;
                    }
                }
            }
            else if (can_ramp_up) {
                ui8_counter_duty_cycle_ramp_down = 0;

                if (++ui8_counter_duty_cycle_ramp_up > ui8_controller_duty_cycle_ramp_up_inverse_step) {
                    ui8_counter_duty_cycle_ramp_up = 0;

                    if (ui8_g_duty_cycle < PWM_DUTY_CYCLE_STARTUP) {
                        ui8_g_duty_cycle = PWM_DUTY_CYCLE_STARTUP;  /* avoid sticky crawl */
                    } else if (ui8_g_duty_cycle < PWM_DUTY_CYCLE_MAX) {
                        uint8_t gap  = (uint8_t)(target_for_ramp - ui8_g_duty_cycle);
                        uint8_t step = (gap >= FAST_GAP_UP) ? STEP_UP_FAST : STEP_UP_SLOW;

                        uint16_t next = (uint16_t)ui8_g_duty_cycle + (uint16_t)step;
                        if (next > target_for_ramp)     next = target_for_ramp;
                        if (next > PWM_DUTY_CYCLE_MAX)  next = PWM_DUTY_CYCLE_MAX;
                        ui8_g_duty_cycle = (uint8_t)next;
                    }
                }
            }
            else if ((ui8_field_weakening_enabled != 0u) && (ui8_g_duty_cycle == PWM_DUTY_CYCLE_MAX)) {
                /* No change needed on duty; allow FW to climb smoothly */
                ui8_counter_duty_cycle_ramp_down = 0;
                if (++ui8_counter_duty_cycle_ramp_up > ui8_controller_duty_cycle_ramp_up_inverse_step) {
                    ui8_counter_duty_cycle_ramp_up = 0;
                    if (ui8_fw_hall_counter_offset < ui8_fw_hall_counter_offset_max) {
                        ui8_fw_hall_counter_offset++;
                    }
                }
            }
            else {
                /* Target met; idle counters */
                ui8_counter_duty_cycle_ramp_up   = 0;
                ui8_counter_duty_cycle_ramp_down = 0;
            }
        }

        Custom throttle response curve thats current based(more throttle = motor go more weeeee)
        i made this curve so i can actually control my speed and not have throttle feel like an on off switch
        it takes advantage of the unused value "ui8_adc_pedal_torque_angle_adj" when u move this up or down, it moves the throttle mid lower or higher, either giving more low end punch or ez modulation. i added an ease on the first 30% of the range to help control bike in a low gear low speed after testing my cuty cycle tweak, the smallest bump gave the old wheelie syndrome,but i got the default poo poo thumb throttle still

        static void apply_throttle(void)
{
	// Read throttle ADC (10 bits after >>2)
	ui16_adc_throttle = (XMC_VADC_GROUP_GetResult(vadc_0_group_1_HW , VADC_POT_RESULT_REG ) & 0x0FFF) >> 2;

	if (ui8_throttle_feature_enabled) {
		// map adc value from 0..255 using calibration
		ui8_throttle_adc_in = map_ui8((uint8_t)(ui16_adc_throttle >> 2),
									  ui8_throttle_min,
								ui8_throttle_max,
								(uint8_t)0,
									  (uint8_t)255);

		// --- Static ease-in on first ~30% (tames bumps, keeps full range) ---Hazza
		{
			uint16_t x = ui8_throttle_adc_in;      // 0..255
			const uint16_t T = 77U;                // ~30% of 255

			// Optional: suppress micro-jitter near idle
			if (x < 3U) x = 0U;

			uint16_t y;
			if (x <= T) {
				// Quadratic ease-in mapped 0..T → 0..T
				y = (uint16_t)((x * x + (T >> 1)) / T);
			} else {
				y = x; // above 30%: untouched
			}

			if (y > 255U) y = 255U;
			ui8_throttle_adc_in = (uint8_t)y;
		}

		// --- One-knob mid-point shaping (0..40 → mid at 30..70%) ---Hazza
		{
			uint8_t k = ui8_adc_pedal_torque_angle_adj; // 0..40 from display
			if (k > 40U) k = 40U;

			uint16_t mid_pct = 30U + ((uint16_t)k * 40U) / 40U;  // 30..70
			uint16_t mid255  = (mid_pct * 255U + 50U) / 100U;    // 77..178

			uint16_t x = ui8_throttle_adc_in; // 0..255
			uint16_t y;
			if (x <= 127U) {
				y = (x * mid255 + 63U) / 127U;
			} else {
				uint16_t top = 255U - mid255;
				y = mid255 + (((x - 128U) * top + 63U) / 127U);
			}
			if (y > 255U) y = 255U;
			ui8_throttle_adc_in = (uint8_t)y;
		}
