// PI helpers
typedef struct { float kp, ki, i; float out_min, out_max; } pi_t;
static inline float pi_step(pi_t p, float err, float dt) {
float u = p->kperr + p->i;
// clamp before integral anti-windup
if (u > p->out_max) u = p->out_max;
if (u < p->out_min) u = p->out_min;
// integral update (back-calc)
p->i += p->ki * err * dt;
if (p->i > p->out_max) p->i = p->out_max;
if (p->i < p->out_min) p->i = p->out_min;
return u;
}
static pi_t pi_d, pi_q;

static void auto_pi_from_RL(float rs, float ld, float lq, float pwm_hz,
float *kp_d, float *ki_d, float *kp_q, float *ki_q) {
// bandwidth ≈ pwm/10 (safe for high L motor), continuous-time PI
const float bw = (pwm_hz * 0.1f) * 2.0f * (float)M_PI; // rad/s
*kp_d = ld * bw;
*ki_d = rs * bw;
*kp_q = lq * bw;
*ki_q = rs * bw;
}

void foc_init(const foc_motor_const_t *mc, const foc_tune_t *tc) {
g_mc = *mc;
g_tc = *tc;
// Auto compute PI if zeros provided
if (g_tc.kp_d == 0 || g_tc.ki_d == 0 || g_tc.kp_q == 0 || g_tc.ki_q == 0) {
auto_pi_from_RL(g_mc.rs, g_mc.ld, g_mc.lq, PWM_FREQ_HZ,
&g_tc.kp_d, &g_tc.ki_d, &g_tc.kp_q, &g_tc.ki_q);
}

pi_d = (pi_t){ .kp=g_tc.kp_d, .ki=g_tc.ki_d, .i=0,
.out_min=-FLT_MAX, .out_max=FLT_MAX };
pi_q = (pi_t){ .kp=g_tc.kp_q, .ki=g_tc.ki_q, .i=0,
.out_min=-FLT_MAX, .out_max=FLT_MAX };

g_st = (foc_state_t){0};
g_st.using_halls = 1;
}

// transforms
static inline void clarke(float ia, float ib, float *a, float *b) {
*a = ia;
b = (ia + 2.0fib) * (1.0f/ sqrtf(3.0f));
}
static inline void park(float a, float b, float s, float c, float d, float q) {
d = ca + sb;
q = -sa + cb;
}
static inline void inv_park(float d, float q, float s, float c, float a, float b) {
a = cd - sq;
b = sd + cq;
}

void foc_fast_isr(float ia, float ib, float vbus,
float *duty_u, float *duty_v, float *duty_w)
{
if (!g_enabled) { *duty_u=*duty_v=*duty_w=0.5f; return; }

// Simple Iq ramp & preload
float iq_cmd = g_iq_target;
float iq_step = g_tc.iq_rate_a_per_s * DT_FAST;
static float iq_ref = 0.0f;
// preload when very low ERPM:
const float erpm = get_erpm();
if (fabsf(erpm) < 50.0f) iq_ref = fmaxf(iq_ref, g_tc.iq_preload);
// ramp
if (iq_ref < iq_cmd) iq_ref = fminf(iq_ref + iq_step, iq_cmd);
else if (iq_ref > iq_cmd) iq_ref = fmaxf(iq_ref - iq_step, iq_cmd);

// Angle source: halls below handover, observer above
float angle;
static float omega_el = 0.0f;
if (halls_valid() && fabsf(erpm) < g_tc.sensorless_handover_erpm) {
angle = halls_interp_angle_rad(); g_st.using_halls = 1;
} else {
extern float observer_step(float v_alpha, float v_beta,
float i_alpha, float i_beta,
float dt, float *omega_el_out);
// We’ll fill v_alpha/beta after dq PI; first estimate with last cycle values:
angle = observer_step(0,0,0,0, DT_FAST, &omega_el); // PLL update will use last values internally
g_st.using_halls = 0;
}

const float s = sinf(angle), c = cosf(angle);

// Clarke
float i_alpha, i_beta;
clarke(ia, ib, &i_alpha, &i_beta);

// Park
float id, iq;
park(i_alpha, i_beta, s, c, &id, &iq);
g_st.id_meas = id; g_st.iq_meas = iq;

// PI current control with cross-decoupling
float err_d = 0.0f - id;
float err_q = iq_ref - iq;

// electrical speed estimate available from observer; safe default from ERPM:
if (g_st.using_halls) omega_el = erpm * (2.0f * (float)M_PI / 60.0f);

float vd = pi_step(&pi_d, err_d, DT_FAST) + ( omega_el * g_mc.lq * iq);
float vq = pi_step(&pi_q, err_q, DT_FAST) + (-omega_el * g_mc.ld * id + omega_el * g_mc.lambda);

// Back to alpha/beta, then SVPWM
float v_alpha, v_beta;
inv_park(vd, vq, s, c, &v_alpha, &v_beta);

// Limit vector magnitude to available bus
float v_lim = (g_tc.v_mod_max > 0 ? g_tc.v_mod_max : 1.0f) * (vbus / sqrtf(3.0f));
float v_mag = sqrtf(v_alphav_alpha + v_betav_beta);
if (v_mag > v_lim) {
float scale = v_lim / (v_mag + 1e-9f);
v_alpha *= scale; v_beta *= scale;
}

svpwm_compute(v_alpha, v_beta, vbus, FOC_OVERMOD_FACTOR, duty_u, duty_v, duty_w);

g_st.v_mod = v_mag / (vbus / sqrtf(3.0f));
g_st.elec_angle = angle;
g_st.elec_omega = omega_el;
g_st.id_cmd = 0.0f;
g_st.iq_cmd = iq_ref;
}

#include "observer.h"
#include "foc.h"
#include <math.h>

static foc_motor_const_t oc_mc;
static observer_cfg_t oc_cfg;
static float oc_angle = 0.0f;
static float pll_i = 0.0f;

void observer_init(const foc_motor_const_t *mc, const observer_cfg_t *cfg){
oc_mc = *mc; oc_cfg = *cfg; oc_angle = 0.0f; pll_i = 0.0f;
}

// Minimal PLL observer (use VESC equations if available in your tree)
float observer_step(float v_alpha, float v_beta,
float i_alpha, float i_beta,
float dt, float omega_el_out)
{
// back-EMF = v - Ri - L*di/dt (di/dt term omitted in this minimal sketch)
float e_alpha = v_alpha - oc_mc.rs * i_alpha;
float e_beta = v_beta - oc_mc.rs * i_beta;

float angle_est = atan2f(e_beta, e_alpha);
// PLL
float err = atan2f(sinf(angle_est - oc_angle), cosf(angle_est - oc_angle));
float omega = oc_cfg.kp * err + pll_i; // rad/s
pll_i += oc_cfg.ki * err * dt;
oc_angle += omega * dt;
// wrap
if (oc_angle > M_PI) oc_angle -= 2.0fM_PI;
if (oc_angle < -M_PI) oc_angle += 2.0fM_PI;

if (omega_el_out) *omega_el_out = omega;
return oc_angle;
}

// ---------- svpwm.c (standard 3-phase, centered) ----------
#include "svpwm.h"
#include <math.h>

void svpwm_compute(float v_alpha, float v_beta, float vbus,
float overmod, float *du, float *dv, float *dw)
{
// Normalize to bus
float k = 2.0f / vbus;
float x = k * v_alpha;
float y = k * (v_alpha * (-0.5f) + v_beta * (sqrtf(3.0f)/2.0f));
float z = k * (v_alpha * (-0.5f) - v_beta * (sqrtf(3.0f)/2.0f));

// Zero-sequence injection
float v_min = fminf(x, fminf(y, z));
float v_max = fmaxf(x, fmaxf(y, z));
float v0 = -0.5f * (v_max + v_min);

float ua = x + v0;
float vb = y + v0;
float wc = z + v0;

// Over-mod clamp
const float lim = 0.5f * (overmod > 0 ? overmod : 1.0f);
if (ua > lim) ua = lim; if (ua < -lim) ua = -lim;
if (vb > lim) vb = lim; if (vb < -lim) vb = -lim;
if (wc > lim) wc = lim; if (wc < -lim) wc = -lim;

*du = ua + 0.5f;
*dv = vb + 0.5f;
*dw = wc + 0.5f;
}
